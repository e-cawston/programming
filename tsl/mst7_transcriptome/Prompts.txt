```{r}
do_DGE_one <- function(s2c_tp, control, condition_label,
                       t2g = NULL,
                       aggregation_column = NULL,
                       out_dir = here("Analysis/Data/DGE")) {
  s2c_tp <- s2c_tp %>%
    mutate(name = droplevels(factor(name))) %>%
    mutate(name = relevel(name, ref = control))
  
  message("      Running: ", condition_label, " vs ", control)
  
  # Try-catch to provide better error messages
  so <- tryCatch({
    sleuth_prep(
      s2c_tp,
      ~ name,
      target_mapping = t2g,
      aggregation_column = aggregation_column,
      transformation_function = function(x) log2(x + 0.5),
      num_cores = 1,
      read_bootstrap_tpm = FALSE,
      extra_bootstrap_summary = FALSE  # Reduce file access further
    )
  }, error = function(e) {
    message("Error in sleuth_prep: ", e$message)
    message("Attempting to clean up and retry...")
    gc()
    closeAllConnections()
    Sys.sleep(1)
    
    # Retry once
    sleuth_prep(
      s2c_tp,
      ~ name,
      target_mapping = t2g,
      aggregation_column = aggregation_column,
      transformation_function = function(x) log2(x + 0.5),
      num_cores = 1,
      read_bootstrap_tpm = FALSE,
      extra_bootstrap_summary = FALSE
    )
  })
  
  so <- sleuth_fit(so, ~ name, 'full')
  so <- sleuth_fit(so, ~ 1, 'reduced')
  so <- sleuth_lrt(so, 'reduced', 'full')
  
  coef_name <- paste0("name", condition_label)
  so <- sleuth_wt(so, coef_name)
  res <- sleuth_results(so, coef_name, 'wt')
  
  tp_str <- as.character(unique(s2c_tp$timepoint))
  exp_str <- as.character(unique(s2c_tp$experiment))
  
  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)
  out_file <- file.path(out_dir,
                        paste0("sleuth_", exp_str, "_", tp_str, "H_",
                               condition_label, "_vs_", control, ".csv"))
  write_csv(res, out_file)
  
  # Force cleanup
  rm(so)
  gc()
  closeAllConnections()
  Sys.sleep(1)  # Increased pause
  
  return(res)
}
```

```{r}
run_experiment <- function(s2c_all,
                           experiment_label,
                           control_name,
                           exclude_from_tests = character(),
                           t2g = NULL,
                           aggregation_column = NULL,
                           out_dir = here("Analysis/Data/DGE")) {

  message(">> Running ", experiment_label, "   (control = ", control_name, ")")

  s2c_exp <- s2c_all %>%
    filter(experiment == experiment_label) %>%
    droplevels()

  stopifnot(control_name %in% s2c_exp$name)

  conditions <- setdiff(levels(droplevels(s2c_exp$name)),
                        c(control_name, exclude_from_tests))

  if (length(conditions) == 0) {
    warning("No conditions remain for ", experiment_label)
    return(NULL)
  }

  for (tp in levels(s2c_exp$timepoint)) {

    message("   Timepoint: ", tp, "H")
    s2c_tp <- s2c_exp %>% filter(timepoint == tp)

    if (nrow(s2c_tp) == 0) next
    if (!(control_name %in% s2c_tp$name)) {
      message("    Control not present at this timepoint -- skipped.")
      next
    }

    for (nm in conditions) {
      if (nm %in% s2c_tp$name) {
        do_DGE_one(s2c_tp,
                   control = control_name,
                   condition_label = nm,
                   t2g = t2g,
                   aggregation_column = aggregation_column,
                   out_dir = out_dir)
      }
    }
    
    # Clean up after each timepoint to release file handles
    gc()
    closeAllConnections()
    Sys.sleep(1)  # Increased pause
  }
  
  # Final cleanup after experiment
  gc()
  closeAllConnections()
  message(">> Completed ", experiment_label)
}
```

```r
run_comparison <- function(s2c,
                           comparison_row,
                           t2g = NULL,
                           aggregation_column = NULL,
                           out_dir = here("results", "dge")) {

  label      <- comparison_row$label
  experiment <- comparison_row$experiment
  control    <- comparison_row$control
  test       <- comparison_row$test

  message(">> Running comparison: ", label)

  # Select samples for this comparison
  if (experiment != "Cross") {
    s2c_sub <- s2c %>%
      filter(experiment == experiment) %>%
      filter(name %in% c(control, test)) %>%
      droplevels()
  } else {
    s2c_sub <- s2c %>%
      filter(name %in% c(control, test)) %>%
      droplevels()
  }

  # Ensure control is reference level
  s2c_sub <- s2c_sub %>%
    mutate(name = factor(name)) %>%
    mutate(name = relevel(name, ref = control))

  # Run per timepoint
  timepoints <- levels(s2c_sub$timepoint)

  results <- map(timepoints, function(tp) {

    s2c_tp <- s2c_sub %>% filter(timepoint == tp)

    if (nrow(s2c_tp) == 0) {
      message("   Timepoint ", tp, "H: no samples, skipping.")
      return(NULL)
    }

    message("   Timepoint ", tp, "H: running sleuth")

    so <- sleuth_prep(
      s2c_tp,
      ~ name,
      target_mapping = t2g,
      aggregation_column = aggregation_column,
      transformation_function = function(x) log2(x + 0.5),
      num_cores = 1,
      read_bootstrap_tpm = FALSE,
      extra_bootstrap_summary = FALSE
    )

    so <- sleuth_fit(so, ~ name, 'full')
    so <- sleuth_fit(so, ~ 1, 'reduced')
    so <- sleuth_lrt(so, 'reduced', 'full')

    coef_name <- paste0("name", test)
    so <- sleuth_wt(so, coef_name)
    res <- sleuth_results(so, coef_name, 'wt')

    # Output path
    dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

    out_file <- file.path(
      out_dir,
      paste0("sleuth_", label, "_", tp, "H.csv")
    )

    write_csv(res, out_file)

    rm(so)
    gc()
    closeAllConnections()

    res
  })

  names(results) <- timepoints
  results
}
```

```r
library(dplyr)
library(readr)

library(stringr)
library(here)

# Extract DEG vector from sleuth results
extract_deg <- function(dge_table,
                        qval_threshold = 0.05,
                        fc_threshold = 0) {

  dge_table %>%
    filter(qval <= qval_threshold) %>%
    filter(abs(b) >= fc_threshold) %>%
    pull(target_id) %>%
    unique()
}

# Load DEG tables for a set of comparisons
load_deg_tables <- function(comparison_labels,
                            dge_dir = here("results", "dge"),
                            qval_threshold = 0.05,
                            fc_threshold = 0) {

  deg_list <- map(comparison_labels, function(label) {

    files <- list.files(
      dge_dir,
      pattern = paste0("sleuth_", label, "_.*H.csv"),
      full.names = TRUE
    )

    if (length(files) == 0) {
      message("No DGE files found for: ", label)
      return(NULL)
    }

    degs <- map(files, function(f) {
      tbl <- read_csv(f, show_col_types = FALSE)
      extract_deg(tbl, qval_threshold, fc_threshold)
    })

    # Combine DEGs across timepoints
    unique(unlist(degs))
  })

  names(deg_list) <- comparison_labels
  deg_list
}

# Generate upset plot
plot_upset <- function(deg_list,
                       label,
                       out_dir = here("results", "figures", "upset")) {

  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

  # Convert list to binary matrix for UpSetR
  all_genes <- unique(unlist(deg_list))

  mat <- map_dfc(names(deg_list), function(nm) {
    tibble(!!nm := all_genes %in% deg_list[[nm]])
  })

  mat <- bind_cols(tibble(gene = all_genes), mat)

  # UpSetR requires a data.frame
  df <- as.data.frame(mat)

  png(file.path(out_dir, paste0("upset_", label, ".png")),
      width = 1800, height = 1200, res = 180)

  upset(df,
        nsets = length(deg_list),
        sets = names(deg_list),
        order.by = "freq",
        keep.order = TRUE)

  dev.off()

  invisible(df)
}
```

# Load sleuth DGE tables and extract signed statistics
load_signed_dge <- function(label,
                            dge_dir = here("results", "dge"),
                            qval_threshold = 0.05) {

  files <- list.files(
    dge_dir,
    pattern = paste0("sleuth_", label, "_.*H.csv"),
    full.names = TRUE
  )

  if (length(files) == 0) {
    message("No DGE files found for: ", label)
    return(NULL)
  }

  # Combine timepoints
  dge <- map_dfr(files, read_csv, show_col_types = FALSE)

  # Keep signed statistics
  dge %>%
    filter(!is.na(b)) %>%        # b = signed effect size
    filter(qval <= qval_threshold) %>%
    select(target_id, b)
}

# Prepare matrix for Rank Products (genes x samples)
prepare_rp_matrix <- function(s2c, tpm_matrix, control, test) {

  samples <- s2c %>%
    filter(name %in% c(control, test)) %>%
    arrange(name) %>%
    pull(sample)

  mat <- tpm_matrix %>%
    select(all_of(samples)) %>%
    as.matrix()

  # Class labels: 0 = control, 1 = test
  cl <- ifelse(colnames(mat) %in% samples[s2c$name == test], 1, 0)

  list(mat = mat, cl = cl)
}

# Run Rank Products
run_rank_products <- function(mat, cl) {

  RP(mat,
     cl = cl,
     logged = FALSE,
     gene.names = rownames(mat))
}

# Extract signed fold-change for RP results
extract_signed_fc <- function(rp_res, signed_dge) {

  rp_tbl <- tibble(
    gene = rp_res$gene.names,
    rp_score = rp_res$RP,
    pfp = rp_res$pfp
  )

  rp_tbl %>%
    left_join(signed_dge, by = c("gene" = "target_id")) %>%
    mutate(log2FC = b) %>%   # signed direction from sleuth
    filter(!is.na(log2FC))
}

# Plot Rank Products with correct directionality
plot_rank_products <- function(rp_tbl,
                               label,
                               out_dir = here("results", "figures", "rank_products")) {

  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

  p <- ggplot(rp_tbl, aes(x = log2FC, y = -log10(pfp))) +
    geom_point(alpha = 0.6) +
    geom_vline(xintercept = 0, linetype = "dashed") +
    theme_minimal() +
    ggtitle(paste("Rank Products:", label)) +
    xlab("Signed log2 Fold Change") +
    ylab("-log10(PFP)")

  out_file <- file.path(out_dir, paste0("rank_products_", label, ".png"))
  ggsave(out_file, p, width = 10, height = 8, dpi = 300)

  invisible(p)
}

# Full pipeline wrapper
run_rank_products_pipeline <- function(s2c,
                                       comparison_row,
                                       tpm_matrix,
                                       dge_dir = here("results", "dge")) {

  label      <- comparison_row$label
  control    <- comparison_row$control
  test       <- comparison_row$test

  message(">> Running Rank Products for: ", label)

  signed_dge <- load_signed_dge(label, dge_dir)

  if (is.null(signed_dge)) {
    message("No signed DGE available for: ", label)
    return(NULL)
  }

  rp_input <- prepare_rp_matrix(s2c, tpm_matrix, control, test)

  rp_res <- run_rank_products(rp_input$mat, rp_input$cl)

  rp_tbl <- extract_signed_fc(rp_res, signed_dge)

  plot_rank_products(rp_tbl, label)

  rp_tbl
}

```r
prepare_comparison_subset <- function(s2c, comparison_row) {
  control <- comparison_row$control
  test <- comparison_row$test
  experiment <- comparison_row$experiment

  if (is.null(experiment) || is.na(experiment) || identical(experiment, "Cross")) {
    s2c_sub <- s2c %>%
      filter(name %in% c(control, test)) %>%
      droplevels()
  } else {
    s2c_sub <- s2c %>%
      filter(experiment == experiment) %>%
      filter(name %in% c(control, test)) %>%
      droplevels()
  }

  s2c_sub %>%
    mutate(name = factor(name),
           timepoint = factor(timepoint)) %>%
    mutate(name = relevel(name, ref = control))
}

standardise_dge_result <- function(res, comparison_row, timepoint) {
  if (is.null(res) || nrow(res) == 0) {
    return(tibble())
  }

  res %>%
    mutate(
      comparison_label = comparison_row$label,
      timepoint = as.character(timepoint),
      control = comparison_row$control,
      test = comparison_row$test,
      comparison = paste(control, "vs", test, sep = "_vs_")
    )
}

run_sleuth_comparison <- function(s2c,
                                  comparison_row,
                                  t2g = NULL,
                                  aggregation_column = NULL,
                                  out_dir = here("results", "dge"),
                                  num_cores = 1) {
  label <- comparison_row$label
  control <- comparison_row$control
  test <- comparison_row$test

  message(">> Running comparison: ", label)

  s2c_sub <- prepare_comparison_subset(s2c, comparison_row)
  timepoints <- levels(s2c_sub$timepoint)

  results <- map(timepoints, function(tp) {
    s2c_tp <- s2c_sub %>% filter(timepoint == tp)

    if (nrow(s2c_tp) == 0) {
      message("   Timepoint ", tp, "H: no samples, skipping.")
      return(tibble())
    }

    message("   Timepoint ", tp, "H: running sleuth")

    so <- sleuth_prep(
      s2c_tp,
      ~ name,
      target_mapping = t2g,
      aggregation_column = aggregation_column,
      transformation_function = function(x) log2(x + 0.5),
      num_cores = num_cores,
      read_bootstrap_tpm = FALSE,
      extra_bootstrap_summary = FALSE
    )

    so <- sleuth_fit(so, ~ name, "full")
    so <- sleuth_fit(so, ~ 1, "reduced")
    so <- sleuth_lrt(so, "reduced", "full")

    coef_name <- paste0("name", test)
    so <- sleuth_wt(so, coef_name)
    res <- sleuth_results(so, coef_name, "wt")
    res <- standardise_dge_result(res, comparison_row, tp)

    dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)
    out_file <- file.path(out_dir, paste0("sleuth_", label, "_", tp, "H.csv"))
    write_csv(res, out_file)

    rm(so)
    gc()
    closeAllConnections()

    res
  })

  bind_rows(results)
}
```

```r
build_tpm_matrix_from_kallisto <- function(s2c_subset) {
  tpm_list <- pmap(s2c_subset %>% select(sample, path), function(sample, path) {
    abundance_file <- ifelse(
      stringr::str_detect(path, "^/"),
      path,
      file.path(here("raw"), path)
    )

    read_tsv(abundance_file, show_col_types = FALSE) %>%
      select(target_id, tpm) %>%
      rename(!!sample := tpm)
  })

  tpm_mat <- reduce(tpm_list, full_join, by = "target_id")
  rownames_mat <- tpm_mat$target_id

  tpm_mat <- tpm_mat %>%
    select(-target_id) %>%
    as.matrix()

  rownames(tpm_mat) <- rownames_mat
  tpm_mat[, s2c_subset$sample, drop = FALSE]
}

run_rank_products_comparison <- function(s2c,
                                         comparison_row,
                                         timepoint = NULL,
                                         out_dir = here("results", "rank_products"),
                                         num_perm = 100) {
  group1 <- comparison_row$control
  group2 <- comparison_row$test
  label <- comparison_row$label

  s2c_sub <- s2c %>%
    filter(name %in% c(group1, group2)) %>%

    return(tibble())
  }

  expr_mat <- build_tpm_matrix_from_kallisto(s2c_sub) %>%
    log2(base = 2, x = . + 0.5)

  n1 <- sum(s2c_sub$name == group1)
  n2 <- sum(s2c_sub$name == group2)

    data = expr_mat,
    cl = cl,
    num.perm = num_perm,
    logged = TRUE,
    gene.names = rownames(expr_mat),
    plot = FALSE
  )

  tg <- topGene(rp_result, cutoff = 1, method = "pfp", logged = TRUE, logbase = 2)

  tidy_rankprod_tbl <- function(tbl, direction, comparison_label, tp) {
    if (is.null(tbl) || nrow(tbl) == 0) {
      return(tibble())
    }

    as.data.frame(tbl) %>%
      tibble::rownames_to_column("target_id") %>%
      mutate(
        direction = direction,
        timepoint = as.character(tp),
        comparison_label = comparison_label,
        group1 = group1,
        group2 = group2
      )
  }

  res_df <- bind_rows(
    tidy_rankprod_tbl(tg$Table1, paste0("up_in_", group1), label, timepoint),
    tidy_rankprod_tbl(tg$Table2, paste0("up_in_", group2), label, timepoint)
  )

  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)
  out_file <- file.path(out_dir, paste0("rankprod_", label, "_", timepoint, "H.csv"))
  write_csv(res_df, out_file)

  res_df
}

run_rank_products_comparisons <- function(s2c,
                                          comparisons,
                                          out_dir = here("results", "rank_products"),
                                          num_perm = 100) {
  results <- map(seq_len(nrow(comparisons)), function(i) {
    comparison_row <- comparisons[i, , drop = FALSE]
    timepoints <- levels(factor(s2c$timepoint))

    map(timepoints, function(tp) {
      run_rank_products_comparison(
        s2c = s2c,
        comparison_row = comparison_row,
        timepoint = tp,
        out_dir = out_dir,
        num_perm = num_perm
      )
    }) %>%
      bind_rows()
  })

  bind_rows(results)
}

plot_rank_products_counts <- function(rp_table,
                                      comparison_label,
                                      out_dir = here("results", "figures", "rank_products"),
                                      pfp_threshold = 0.05) {
  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

  comparison_df <- rp_table %>%
    filter(comparison_label == !!comparison_label) %>%
    filter(pfp <= pfp_threshold)

  if (nrow(comparison_df) == 0) {
    return(NULL)
  }

  group1 <- unique(comparison_df$group1)
  group2 <- unique(comparison_df$group2)

  plot_df <- comparison_df %>%
    mutate(
      direction_label = if_else(direction == paste0("up_in_", group1), "up", "down"),
      n_plot = if_else(direction_label == "down", -n(), n()),
      timepoint_num = suppressWarnings(as.numeric(str_remove(timepoint, "H")))
    ) %>%
    count(timepoint, direction_label, name = "n") %>%
    mutate(
      n_plot = if_else(direction_label == "down", -n, n),
      timepoint = forcats::fct_reorder(timepoint, timepoint_num, .desc = FALSE)
    )

  p <- ggplot(plot_df, aes(x = n_plot, y = timepoint, fill = direction_label)) +
    geom_col(width = 0.7) +
    geom_vline(xintercept = 0, linewidth = 0.5) +
    scale_fill_manual(
      values = c(up = "#D62728", down = "#1F77B4"),
      labels = c(up = paste("Higher in", group1), down = paste("Higher in", group2))
    ) +
    scale_x_continuous(labels = abs) +
    labs(
      title = paste("Rank products:", comparison_label),
      subtitle = paste0("pfp < ", pfp_threshold),
      x = "Number of DEGs",
      y = "Timepoint",
      fill = NULL
    ) +
    theme_classic(base_size = 12)

  out_file <- file.path(out_dir, paste0("rankprod_", comparison_label, ".png"))
  ggsave(out_file, p, width = 8, height = 5, dpi = 300)

  p
}
```
